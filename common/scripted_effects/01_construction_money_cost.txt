# =============================================================================
# MD-Realistic-2000-Rebalance — IC-Proportional Construction Money Cost (v1.1.4)
# =============================================================================
# Charges money to treasury proportional to IC being spent on active construction.
#
# v1.1.4 FIXES (vs v1.1.3):
#   1. Uses gdpc_converging_var instead of gdp_per_capita
#      - gdp_per_capita is overwritten every cycle by MD's calculate_gdp
#      - gdpc_converging_var is smoothed/stable, doesn't fluctuate per cycle
#
#   2. Direct treasury drain (subtract_from_variable = treasury) instead of
#      additional_expenses_rate which MD's calculate_additional_expense_rate
#      RESETS to 0 each monthly cycle (line 3119 of MD's 00_money_system.txt)
#
#   3. Construction detection: tries is_building_constructing first, falls back
#      to civilian_factories_manpower_fulfillment > 0.5 as proxy
#
# FORMULA:
#   weekly_drain = active_civs × 31.5 IC × $0.20M × cost_factor
#
# Where:
#   active_civs   = industrial_complex_total × civilian_factories_manpower_fulfillment
#   31.5 IC/week  = 4.5 IC/day × 7 days
#   $0.20M per IC = calibration constant ($0.0002B per IC)
#   cost_factor   = 0.30 + 0.014 × gdpc_converging_var (clamped 0.30-1.50)
#
# EXAMPLES (cost_factor at typical convergence values):
#   USA (gdpc ~36): factor 0.81  → industrial_complex (33150 IC) total ~$5.4B
#   Polska (gdpc ~6): factor 0.38 → industrial_complex total ~$2.5B
#   Russia (gdpc ~8): factor 0.41 → industrial_complex total ~$2.7B
#   Indie (gdpc ~2): factor 0.33  → industrial_complex total ~$2.2B
#
# TUNING:
#   Lower md_dollars_per_ic for cheaper construction (default 0.0002 = $200k/IC)
# =============================================================================


# -----------------------------------------------------------------------------
# Effect: calculate_construction_money_cost
# Called: weekly via on_actions
# Purpose: Drain treasury proportional to IC spent on construction
# -----------------------------------------------------------------------------
calculate_construction_money_cost = {

	# Step 1: Compute cost factor using SMOOTHED gdp/c (not raw)
	# gdpc_converging_var is MD's stable proxy for gdp_per_capita
	# It's used internally in MD's calculations to avoid spikes
	set_temp_variable = { md_cost_factor = gdpc_converging_var }
	multiply_temp_variable = { md_cost_factor = 0.014 }
	add_to_temp_variable = { md_cost_factor = 0.30 }
	clamp_temp_variable = { var = md_cost_factor min = 0.30 max = 1.50 }

	# Save for tooltip/display
	set_variable = { md_realistic_cost_factor_display = md_cost_factor }

	# Step 2: Detect active construction with fallback
	#   Primary:  any owned state with is_building_constructing = yes
	#   Fallback: civilian_factories_manpower_fulfillment > 0.5
	#             (means civs are working, likely building rather than producing CG)
	set_variable = { md_any_construction = 0 }

	# Try primary detection
	every_owned_state = {
		limit = { is_building_constructing = yes }
		set_variable = { PREV.md_any_construction = 1 }
	}

	# Fallback: if primary detected nothing but civs are highly utilized,
	# assume they're constructing
	if = {
		limit = {
			check_variable = { md_any_construction = 0 }
			check_variable = { civilian_factories_manpower_fulfillment > 0.5 }
			check_variable = { industrial_complex_total > 0 }
		}
		set_variable = { md_any_construction = 1 }
	}

	# Reset cost variable each tick
	set_variable = { md_realistic_construction_cost = 0 }

	# Step 3: Calculate IC-based weekly drain
	if = {
		limit = { check_variable = { md_any_construction > 0 } }

		set_temp_variable = { md_active_civs = industrial_complex_total }
		multiply_temp_variable = { md_active_civs = civilian_factories_manpower_fulfillment }

		# IC per week (4.5 IC/day × 7 days = 31.5 IC/week per civ)
		set_temp_variable = { md_ic_per_week = md_active_civs }
		multiply_temp_variable = { md_ic_per_week = 31.5 }

		# Dollars per IC ($0.0002B = $200k)
		set_temp_variable = { md_dollars_per_ic = 0.0002 }

		# Weekly drain in BILLIONS USD
		set_variable = { md_realistic_construction_cost = md_ic_per_week }
		multiply_variable = { md_realistic_construction_cost = md_dollars_per_ic }
		multiply_variable = { md_realistic_construction_cost = md_cost_factor }

		# Clamp to sane range (no negative drain, max $5B/week to prevent runaway)
		clamp_variable = { var = md_realistic_construction_cost min = 0 max = 5 }
	}

	# Step 4: DIRECT treasury drain (weekly)
	# This bypasses MD's calculate_additional_expense_rate which would reset
	# any addition we make. Treasury decreases visibly each week.
	subtract_from_variable = { treasury = md_realistic_construction_cost }
}


# -----------------------------------------------------------------------------
# Effect: md_realistic_init_variables
# Called: once at game start
# -----------------------------------------------------------------------------
md_realistic_init_variables = {
	set_variable = { md_realistic_construction_cost = 0 }
	set_variable = { md_realistic_cost_factor_display = 0.30 }
	set_variable = { md_any_construction = 0 }
}
