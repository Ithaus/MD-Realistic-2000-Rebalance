# =============================================================================
# MD-Realistic-2000-Rebalance — IC-Proportional Construction Money Cost
# =============================================================================
# Charges money to treasury proportional to the IC (Industrial Capacity) being
# spent on construction.
#
# FORMULA:
#   monthly_drain = active_civs × 135 IC/month × $0.20M × cost_factor
#   where:
#     active_civs    = industrial_complex_total × civilian_factories_manpower_fulfillment
#     135 IC/month   = 4.5 IC/day × 30 days (default civ output rate)
#     $0.20M per IC  = calibration constant (md_dollars_per_ic = 0.0002 in $B)
#     cost_factor    = 0.30 + 0.014 × gdp_per_capita (clamped 0.30-1.50)
#
# EXAMPLES OF TOTAL BUILDING COSTS (USA factor 0.81):
#   industrial_complex (33,150 IC) → $5.37B
#   arms_factory       (42,000 IC) → $6.80B
#   dockyard           (45,000 IC) → $7.29B
#   offices            (40,000 IC) → $6.48B
#   nuclear_reactor    (50,000 IC) → $8.10B
#   infrastructure     (10,000 IC) → $1.62B
#
# IMPLEMENTATION NOTES:
#   - Hook is on_MONTHLY (not weekly). MD's tax engine resets
#     additional_expenses_rate every month, so weekly additions get overwritten.
#   - We feed our drain into additional_expenses_rate which appears in MD's
#     budget panel under "Other Expenses".
#   - We do NOT directly subtract from treasury — MD's existing system handles
#     that based on additional_expenses_rate.
#
# TUNING:
#   md_dollars_per_ic = 0.0002 (default $200k per IC)
#     Lower (0.0001) = 50% cheaper construction
#     Higher (0.0004) = 200% more expensive
# =============================================================================


# -----------------------------------------------------------------------------
# Effect: calculate_construction_money_cost
# Called: monthly via on_actions
# Purpose: Compute monthly drain from active construction; feed into MD's budget
# -----------------------------------------------------------------------------
calculate_construction_money_cost = {
	# Step 1: Country-specific cost factor (richer = more expensive)
	# Formula: 0.30 + 0.014 × gdp_per_capita (in thousands USD)
	# Examples: USA $36k → 0.81, Niemcy $24k → 0.63, Polska $4.5k → 0.36, Indie $0.4k → 0.31
	set_temp_variable = { md_cost_factor = gdp_per_capita }
	multiply_temp_variable = { md_cost_factor = 0.014 }
	add_to_temp_variable = { md_cost_factor = 0.30 }
	clamp_temp_variable = { var = md_cost_factor min = 0.30 max = 1.50 }

	# Save for tooltip display
	set_variable = { md_realistic_cost_factor_display = md_cost_factor }

	# Step 2: Check if any state is constructing something
	# Only charge money if there's actually active construction
	set_variable = { md_any_construction = 0 }
	every_owned_state = {
		limit = { is_building_constructing = yes }
		set_variable = { PREV.md_any_construction = 1 }
	}

	# Reset main cost variable
	set_variable = { md_realistic_construction_cost = 0 }

	# Step 3: If building anything, calculate IC-based monthly drain
	if = {
		limit = { check_variable = { md_any_construction > 0 } }

		# Active civilian factories engaged in construction
		# Use full count × manpower fulfillment (NO 50% reduction —
		# we assume player dedicates civs to queue when building)
		set_temp_variable = { md_active_civs = industrial_complex_total }
		multiply_temp_variable = { md_active_civs = civilian_factories_manpower_fulfillment }

		# IC per month from those civs
		# Default: 4.5 IC/day × 30 days = 135 IC/month per civ
		set_temp_variable = { md_ic_per_month = md_active_civs }
		multiply_temp_variable = { md_ic_per_month = 135 }

		# Dollars per IC point (calibration constant)
		# 0.0002 = $200k per IC = $200M per 1,000 IC
		set_temp_variable = { md_dollars_per_ic = 0.0002 }

		# Monthly drain in BILLIONS USD
		# = ic_per_month × $/IC × cost_factor
		set_variable = { md_realistic_construction_cost = md_ic_per_month }
		multiply_variable = { md_realistic_construction_cost = md_dollars_per_ic }
		multiply_variable = { md_realistic_construction_cost = md_cost_factor }
	}

	# Step 4: Feed into MD's existing expense system
	# additional_expenses_rate appears in budget panel under "Other Expenses"
	add_to_variable = { additional_expenses_rate = md_realistic_construction_cost }
}


# -----------------------------------------------------------------------------
# Effect: md_realistic_init_variables
# Called: once at game start via on_startup
# -----------------------------------------------------------------------------
md_realistic_init_variables = {
	set_variable = { md_realistic_construction_cost = 0 }
	set_variable = { md_realistic_cost_factor_display = 0.30 }
	set_variable = { md_any_construction = 0 }
}
