# =============================================================================
# MD-Realistic-2000-Rebalance — on_actions hooks
# =============================================================================
# - on_startup: initialize tracking variables, add informational idea
# - on_monthly: compute construction money cost (feeds into budget panel)
# =============================================================================

on_actions = {

	on_startup = {
		effect = {
			every_country = {
				limit = { exists = yes }
				md_realistic_init_variables = yes
			}
		}
	}

	on_monthly = {
		effect = {
			every_country = {
				limit = { exists = yes }
				calculate_construction_money_cost = yes
			}
		}
	}
}
