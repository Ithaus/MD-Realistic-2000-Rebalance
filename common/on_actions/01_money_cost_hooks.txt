# =============================================================================
# MD-Realistic-2000-Rebalance — on_actions hooks (v1.1.4)
# =============================================================================
# Hook is on_WEEKLY (changed from on_monthly in v1.1.4) for:
# - Granular treasury drain (player sees treasury decrease each week)
# - More responsive feedback when starting/finishing construction
# =============================================================================

on_actions = {

	on_startup = {
		effect = {
			every_country = {
				limit = { exists = yes }
				md_realistic_init_variables = yes
			}
		}
	}

	on_weekly = {
		effect = {
			every_country = {
				limit = { exists = yes }
				calculate_construction_money_cost = yes
			}
		}
	}
}
